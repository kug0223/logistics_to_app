## 신뢰성 기반 지원 시스템

### 1. 기본 설계

#### 문제 정의
```
시나리오:
지원자 A가 같은 날 여러 공고에 지원
- 신세계 (06:00~14:00) - 대기중
- 쿠팡 (09:00~18:00) - 대기중
- 마켓컬리 (13:00~22:00) - 대기중

쿠팡이 확정되면?
→ 신세계, 마켓컬리와 시간 겹침!
→ 두 지원을 어떻게 처리?
```

#### 해결 방안
**확정 시 자동 취소 + 알림 발송**

### 2. 확정 시 자동 취소 로직

#### 2.1 핵심 함수: `confirmApplication()`

```dart
/// 지원서 확정 + 충돌하는 지원 자동 취소
Future<void> confirmApplication(String appId, String adminUID) async {
  // 1. 확정할 지원서 조회
  final app = await getApplication(appId);
  
  // 2. 같은 날짜, 시간 겹치는 대기중 지원서 찾기
  final conflictingApps = await findConflictingApplications(
    uid: app.uid,
    workDate: app.workDate,
    startTime: app.startTime,
    endTime: app.endTime,
    excludeId: appId,
    status: 'PENDING',  // 대기중만
  );
  
  // 3. 배치로 처리 (트랜잭션)
  final batch = FirebaseFirestore.instance.batch();
  
  // 3-1. 확정 처리
  batch.update(
    _firestore.collection('applications').doc(appId),
    {
      'status': 'CONFIRMED',
      'confirmedAt': FieldValue.serverTimestamp(),
      'confirmedBy': adminUID,
    },
  );
  
  // 3-2. 충돌하는 지원들 자동 취소
  for (var conflictApp in conflictingApps) {
    batch.update(
      _firestore.collection('applications').doc(conflictApp.id),
      {
        'status': 'AUTO_CANCELED',  // 새로운 상태!
        'canceledAt': FieldValue.serverTimestamp(),
        'cancelReason': 'SCHEDULE_CONFLICT',
        'conflictingAppId': appId,
        'conflictingBusiness': app.businessName,
        'conflictingTime': '${app.startTime}~${app.endTime}',
      },
    );
    
    // TO의 통계 업데이트 (pendingCount -1)
    await _updateTOPendingCount(conflictApp, -1);
  }
  
  await batch.commit();
  
  // 4. 알림 발송
  if (conflictingApps.isNotEmpty) {
    await _sendCancelNotifications(
      userId: app.uid,
      canceledApps: conflictingApps,
      confirmedApp: app,
    );
  }
  
  print('✅ 확정 완료 + ${conflictingApps.length}개 자동 취소');
}
```

#### 2.2 충돌 지원서 찾기

```dart
/// 시간이 겹치는 지원서 찾기
Future<List<ApplicationModel>> findConflictingApplications({
  required String uid,
  required DateTime workDate,
  required String startTime,
  required String endTime,
  required String excludeId,
  String status = 'PENDING',
}) async {
  // 1. 같은 날짜의 모든 지원서 조회
  final snapshot = await _firestore
      .collection('applications')
      .where('uid', isEqualTo: uid)
      .where('workDate', isEqualTo: Timestamp.fromDate(workDate))
      .where('status', isEqualTo: status)
      .get();
  
  // 2. 시간대 겹침 필터링
  final conflicts = <ApplicationModel>[];
  
  for (var doc in snapshot.docs) {
    if (doc.id == excludeId) continue;
    
    final app = ApplicationModel.fromFirestore(doc);
    
    // 시간 겹침 체크
    if (_hasTimeOverlap(startTime, endTime, app.startTime, app.endTime)) {
      conflicts.add(app);
    }
  }
  
  return conflicts;
}

/// 두 시간대가 겹치는지 체크
bool _hasTimeOverlap(String s1, String e1, String s2, String e2) {
  final start1 = _timeToMinutes(s1);
  final end1 = _timeToMinutes(e1);
  final start2 = _timeToMinutes(s2);
  final end2 = _timeToMinutes(e2);
  
  // 겹치지 않는 경우: end1 <= start2 || end2 <= start1
  // 겹치는 경우: 위의 반대
  return !(end1 <= start2 || end2 <= start1);
}

/// 시간을 분 단위로 변환 (예: "09:30" → 570)
int _timeToMinutes(String time) {
  final parts = time.split(':');
  return int.parse(parts[0]) * 60 + int.parse(parts[1]);
}
```

### 3. 지원 시 충돌 체크

#### 3.1 핵심 함수: `applyForTO()`

```dart
/// TO 지원하기 (충돌 체크 포함)
Future<bool> applyForTO({
  required String uid,
  required TOModel to,
  required WorkDetailModel workDetail,
}) async {
  // 1. 이미 확정된 근무 조회
  final confirmedSchedules = await getConfirmedSchedules(
    uid: uid,
    workDate: to.date,
  );
  
  // 2. 시간대 충돌 체크
  for (var schedule in confirmedSchedules) {
    if (_hasTimeOverlap(
      workDetail.startTime,
      workDetail.endTime,
      schedule.startTime,
      schedule.endTime,
    )) {
      // ⚠️ 충돌 발견!
      throw Exception(
        '이미 ${schedule.startTime}~${schedule.endTime}에\n'
        '${schedule.businessName}에서 확정된 근무가 있습니다.\n\n'
        '해당 시간대에는 추가 지원이 불가능합니다.'
      );
    }
  }
  
  // 3. 충돌 없으면 지원 생성
  await _createApplication(...);
  return true;
}

/// 확정된 근무 일정 조회
Future<List<ApplicationModel>> getConfirmedSchedules({
  required String uid,
  required DateTime workDate,
}) async {
  final snapshot = await _firestore
      .collection('applications')
      .where('uid', isEqualTo: uid)
      .where('workDate', isEqualTo: Timestamp.fromDate(workDate))
      .where('status', isEqualTo: 'CONFIRMED')
      .get();
  
  return snapshot.docs
      .map((doc) => ApplicationModel.fromFirestore(doc))
      .toList();
}
```

```dart
/// 자동 취소 알림 발송
Future<void> _sendCancelNotifications({
  required String userId,
  required List<ApplicationModel> canceledApps,
  required ApplicationModel confirmedApp,
}) async {
  for (var app in canceledApps) {
    // 4-1. FCM 푸시 알림 (나중에 구현)
    await _sendPushNotification(
      userId: userId,
      title: '지원이 자동 취소되었습니다',
      body: '${confirmedApp.businessName} 확정으로 인해\n'
            '${app.businessName} 지원이 취소되었습니다.\n'
            '시간: ${app.startTime}~${app.endTime}',
      data: {
        'type': 'AUTO_CANCEL',
        'canceledAppId': app.id,
        'confirmedAppId': confirmedApp.id,
      },
    );
    
    // 4-2. 인앱 알림 저장 (즉시 구현)
    await _createInAppNotification(
      userId: userId,
      type: 'AUTO_CANCEL',
      title: '지원 자동 취소',
      message: '${app.businessName} (${app.workDate.month}/${app.workDate.day} '
               '${app.startTime}~${app.endTime}) 지원이 자동 취소되었습니다.\n\n'
               '이유: ${confirmedApp.businessName}에서 확정됨',
      relatedAppId: app.id,
      createdAt: DateTime.now(),
    );
  }
}

/// 인앱 알림 생성
Future<void> _createInAppNotification({
  required String userId,
  required String type,
  required String title,
  required String message,
  String? relatedAppId,
  required DateTime createdAt,
}) async {
  await _firestore.collection('notifications').add({
    'userId': userId,
    'type': type,
    'title': title,
    'message': message,
    'relatedAppId': relatedAppId,
    'isRead': false,
    'createdAt': Timestamp.fromDate(createdAt),
  });
}
```

```dart
    required this.endTime,

      'confirmedAt': confirmedAt != null ? Timestamp.fromDate(confirmedAt!) : null,
      'confirmedBy': confirmedBy,
    };
```

## 연속 근무 간격 관리

### 1. 문제 정의

#### 시나리오
```
확정 1: 신세계푸드 송파 (09:00~12:00)
지원 시도: 쿠팡 김포 (12:30~15:00)

문제:
- 송파 → 김포: 차로 1시간
- 12:00 퇴근 → 12:30 출근 불가능!
- 노쇼 발생 가능성 ↑
```

#### 해결 방향
**지원자가 직접 설정한 최소 휴식 시간 적용**

```dart
  final DateTime? lastNoShowDate;

  final DateTime? suspendedUntil;
  final String? suspensionReason;

    required this.uid,
    this.minRestMinutes = 60,  // 기본값: 60분
    this.noShowCount = 0,
    this.lastNoShowDate,
    this.reliability = 100.0,
    this.isSuspended = false,
    this.suspendedUntil,
    this.suspensionReason,
  });
  
  // Firestore 변환
  factory UserSettings.fromMap(Map<String, dynamic> data, String uid) {
    return UserSettings(
      uid: uid,
      minRestMinutes: data['minRestMinutes'] ?? 60,
```

```dart
Future<List<JobRecommendation>> getSmartRecommendations({

  // 1. 해당 날짜의 모든 공고 조회

    return !_hasTimeConflict(confirmedApp, job);
  }).toList();
  
  // 3. 점수 계산 및 추천 생성
  final recommendations = <JobRecommendation>[];
  
  for (var job in filtered) {
    // 시간 점수
    final timeResult = _calculateTimeScore(confirmedApp, job);
    
    // 거리 점수
    final distanceResult = _calculateDistanceScore(confirmedApp, job);
    
    // 급여 점수

    // 신뢰도 점수

    // 총점 계산 (가중 평균)
    final totalScore = 
      timeResult.score * 0.4 +
      distanceResult.score * 0.3 +
      wageScore * 0.2 +
      reliabilityScore * 0.1;
    
    // 추천 객체 생성
    recommendations.add(JobRecommendation(
      to: job.to,
      workDetail: job.workDetail,
      score: totalScore,
      timeScore: timeResult.score,
      distanceScore: distanceResult.score,
      wageScore: wageScore,
      reliabilityScore: reliabilityScore,
      distance: distanceResult.distance,
      travelTime: distanceResult.travelTime,
      gapMinutes: timeResult.gapMinutes,
      estimatedTotalWage: confirmedApp.wage + job.workDetail.wage,
    ));
  }
  
  // 4. 정렬: 점수 높은 순
  recommendations.sort((a, b) => b.score.compareTo(a.score));
```

```dart
/// 시간 적합도 계산
TimeScoreResult _calculateTimeScore(
  ApplicationModel confirmed,
  AvailableJob newJob,
) {
  // 확정 근무 종료 시간
  final confirmedEnd = _timeToMinutes(confirmed.endTime);
  
  // 새 근무 시작 시간
  final newStart = _timeToMinutes(newJob.workDetail.startTime);
  
  // 간격 (분)
  final gap = newStart - confirmedEnd;
  
  // 점수 부여
  double score;
  
  if (gap < 30) {
    score = 0;  // 30분 미만: 불가능
  } else if (gap >= 30 && gap <= 60) {
    score = 100;  // 30~60분: 최적!
  } else if (gap > 60 && gap <= 90) {
    score = 85;   // 60~90분: 좋음
  } else if (gap > 90 && gap <= 120) {
    score = 70;   // 90~120분: 괜찮음
  } else if (gap > 120 && gap <= 180) {
    score = 50;   // 2~3시간: 시간 낭비
  } else {
    score = 30;   // 3시간 이상: 비효율
  }
  
  return TimeScoreResult(score, gap);
}
```

```dart
class DistanceScoreResult {
  final double score;
  final double distance;      // km
  final int travelTime;       // 분
  
  DistanceScoreResult(this.score, this.distance, this.travelTime);
}

/// 거리 점수 계산
DistanceScoreResult _calculateDistanceScore(
  ApplicationModel confirmed,
  AvailableJob newJob,
) {
  // GPS 좌표로 거리 계산 (Haversine formula)
  final distance = _calculateDistance(
    confirmed.businessLocation,
    newJob.to.location,
  );
  
  // 예상 이동 시간 (분)
  // 간단한 추정: 5km/h 도보, 20km/h 대중교통
  int travelTime;
  if (distance <= 1) {
    travelTime = (distance * 12).toInt();  // 도보 (5km/h)
  } else {
    travelTime = (distance * 3).toInt();   // 대중교통 (20km/h)
  }
  
  // 점수 부여
  double score;
  
  if (distance <= 1) {
    score = 100;  // 1km 이내: 도보권
  } else if (distance <= 3) {
    score = 85;   // 3km 이내: 버스/자전거
  } else if (distance <= 5) {
    score = 70;   // 5km 이내: 버스
  } else if (distance <= 10) {
    score = 50;   // 10km 이내: 지하철
  } else if (distance <= 15) {
    score = 30;   // 15km 이내: 먼 거리
  } else {
    score = 10;   // 15km 이상: 비효율
  }
  
  return DistanceScoreResult(score, distance, travelTime);
}

/// Haversine 공식으로 두 GPS 좌표 사이 거리 계산
double _calculateDistance(LatLng from, LatLng to) {
  const R = 6371; // 지구 반지름 (km)
  
  final dLat = _toRadians(to.latitude - from.latitude);
  final dLon = _toRadians(to.longitude - from.longitude);
  
  final a = sin(dLat / 2) * sin(dLat / 2) +
      cos(_toRadians(from.latitude)) * 
      cos(_toRadians(to.latitude)) *
      sin(dLon / 2) * sin(dLon / 2);
  
  final c = 2 * atan2(sqrt(a), sqrt(1 - a));
  
  return R * c;
}

double _toRadians(double degrees) {
  return degrees * pi / 180;
}
```

```dart
  } else if (hourlyWage >= 9860) {

  }

/// 근무 시간 계산 (시간 단위)
double _calculateHours(String startTime, String endTime) {
  final start = _timeToMinutes(startTime);
  final end = _timeToMinutes(endTime);
  return (end - start) / 60.0;
}
```

```dart
Future<double> _calculateReliabilityScore(String businessId) async {
  // 사업장 정보 조회
  final business = await _getBusinessInfo(businessId);
  
  if (business == null) return 50;  // 정보 없으면 중간 점수
  
  // 평점 (0~5)
  final rating = business.averageRating ?? 3.0;
  
  // 확정률 (0~100)
  final confirmRate = business.confirmationRate ?? 50.0;
  
  // 점수 계산
  final ratingScore = (rating / 5.0) * 50;  // 최대 50점
  final confirmScore = (confirmRate / 100.0) * 50;  // 최대 50점
  
  return ratingScore + confirmScore;
}
```

```dart
class RecommendationCache {
  static final Map<String, List<JobRecommendation>> _cache = {};
  
  static String _getCacheKey(String uid, DateTime date) {
    return '${uid}_${date.year}${date.month}${date.day}';
  }
  
  static List<JobRecommendation>? get(String uid, DateTime date) {
    final key = _getCacheKey(uid, date);
    return _cache[key];
  }
  
  static void set(String uid, DateTime date, List<JobRecommendation> recs) {
    final key = _getCacheKey(uid, date);
    _cache[key] = recs;
    
    // 1시간 후 자동 삭제
    Future.delayed(Duration(hours: 1), () {
      _cache.remove(key);
    });
  }
  
  static void clear() {
    _cache.clear();
  }
}
```

```
   ✅ 쿠팡 확정됨
```

```
   ✅ 에러 다이얼로그 표시
   ✅ 지원 불가능
```
